async function scanAllMarketBy24hChange() { try { console.log("🔄 Fetching 24hr ticker data for all Binance spot coins..."); // Binance ki 24hr ticker API se saare coins ka data aik sath aa jata hai const res = await fetch('https://data-api.binance.vision/api/v3/ticker/24hr'); const tickers = await res.json(); // Sirf USDT spot trading pairs filter karna (Leverage tokens skip karke) const usdtTickers = tickers.filter(t => t.symbol.endsWith('USDT') && !t.symbol.includes('UP') && !t.symbol.includes('DOWN') ); let uptrendCount = 0; let downtrendCount = 0; let sidewaysCount = 0; let totalChecked = usdtTickers.length; usdtTickers.forEach(t => { const changePercent = parseFloat(t.priceChangePercent); // Simple Logic: // Agar 24h change +1.5% se zyada hai -> Uptrend // Agar 24h change -1.5% se kam hai -> Downtrend // Agar beech mein hai -> Sideways if (changePercent > 1.5) { uptrendCount++; } else if (changePercent < -1.5) { downtrendCount++; } else { sidewaysCount++; } }); // Percentages calculate karna const upPercent = ((uptrendCount / totalChecked) * 100).toFixed(1); const downPercent = ((downtrendCount / totalChecked) * 100).toFixed(1); const sidewaysPercent = ((sidewaysCount / totalChecked) * 100).toFixed(1); // Overall Market Decision let marketSentiment = ""; if (upPercent >= 60) { marketSentiment = "🚀 STRONG BULLISH MARKET (Uptrend Dominant!)"; } else if (downPercent >= 60) { marketSentiment = "🩸 STRONG BEARISH MARKET (Downtrend Dominant!)"; } else { marketSentiment = "⚖️ SIDEWAYS / MIXED MARKET (Ranging Condition)"; } // Print Complete Summary console.log("\n=================================================="); console.log(" 🌐 BINANCE 24H MARKET BREADTH REPORT "); console.log("=================================================="); console.log(`📊 Total Coins Scanned : ${totalChecked}`); console.log("--------------------------------------------------"); console.log(`🟢 Uptrend Coins : ${upPercent}% (${uptrendCount} coins)`); console.log(`🔴 Downtrend Coins : ${downPercent}% (${downtrendCount} coins)`); console.log(`🟡 Sideways Coins : ${sidewaysPercent}% (${sidewaysCount} coins)`); console.log("=================================================="); console.log(`🎯 MARKET STATUS : ${marketSentiment}`); console.log("==================================================\n"); } catch (err) { console.error("❌ Error fetching data:", err.message); } } // Run Scanner scanAllMarketBy24hChange();
